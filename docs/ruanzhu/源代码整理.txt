function inlineImages(html, baseDir, files) {
  // 同时处理 <img> 与 SVG <image>，兼容双/单引号，支持 src / href / xlink:href
  const tagRe = /<(img|image)\b([^>]*?)\/?>/gi;
  let result = '';
  let last = 0;
  let m;
  while ((m = tagRe.exec(html))) {
    const full = m[0];
    const inner = m[2];
    const attrRe = /\b(src|xlink:href|href)\s*=\s*("([^"]*)"|'([^']*)')/i;
    const am = inner.match(attrRe);
    let replacement = full;
    if (am) {
      const val = am[3] !== undefined ? am[3] : am[4]; // 双引号组3 / 单引号组4
      if (val && !/^(https?:|data:)/i.test(val)) {
        const rel = val.split('#')[0]; // 去掉可能的 fragment
        const abs = normalizePath(joinPath(baseDir, rel));
        const data = files[abs];
        if (data) {
          const mime = mimeFromName(val);
          const b64 = bytesToBase64(data);
          const newUrl = 'data:' + mime + ';base64,' + b64;
          // 替换该属性值，保留原属性名与双引号
          replacement = full.replace(am[0], am[1] + '="' + newUrl + '"');
        }
      }
    }
    result += html.slice(last, m.index) + replacement;
    last = m.index + full.length;
  }
  result += html.slice(last);
  return result;
}
function joinPath(base, rel) {
  if (/^(https?:|data:)/i.test(rel)) return rel;
  const b = base.replace(/\/$/, '');
  let parts = (b ? b.split('/') : []).concat(rel.split('/'));
  const stack = [];
  for (const part of parts) {
    if (part === '' || part === '.') continue;
    if (part === '..') stack.pop();
    else stack.push(part);
  }
  return stack.join('/');
}

// ---------- Markdown 渲染（极简、安全：先转义再处理） ----------
export function renderMarkdown(md) {
  const esc = String(md || '')
    .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
  const lines = esc.split(/\n/);
  let html = '';
  let inList = false;
  const closeList = () => { if (inList) { html += '</ul>'; inList = false; } };
  for (const line of lines) {
    const t = line.replace(/\s+$/, '');
    if (/^###\s+/.test(t)) { closeList(); html += '<h4>' + inline(t.replace(/^###\s+/, '')) + '</h4>'; }
    else if (/^##\s+/.test(t)) { closeList(); html += '<h3>' + inline(t.replace(/^##\s+/, '')) + '</h3>'; }
    else if (/^#\s+/.test(t)) { closeList(); html += '<h2>' + inline(t.replace(/^#\s+/, '')) + '</h2>'; }
    else if (/^[-*]\s+/.test(t)) {
      if (!inList) { html += '<ul>'; inList = true; }
      html += '<li>' + inline(t.replace(/^[-*]\s+/, '')) + '</li>';
    } else if (t === '') { closeList(); }
    else { closeList(); html += '<p>' + inline(t) + '</p>'; }
  }
  closeList();
  return html;
}
function inline(s) {
  return s.replace(/\*\*(.+?)\*\*/g, '<b>$1</b>').replace(/\*(.+?)\*/g, '<i>$1</i>');
}

function parseJson(s) {

------ 第 86 页 / 共 98 页 ------
  if (typeof s !== 'string') return s;
  let t = s.trim();
  // 去掉 ```json ... ``` 或 ``` ... ``` 包裹
  const fence = t.match(/```(?:json)?\s*([\s\S]*?)```/i);
  if (fence) t = fence[1].trim();
  // 截取首个 { 或 [ 到匹配闭合处
  const iBrace = t.indexOf('{');
  const iBrack = t.indexOf('[');
  let start = -1;
  let open = null;
  if (iBrace === -1 && iBrack === -1) {
    throw new Error('AI 未返回可解析的 JSON');
  } else if (iBrace === -1) {
    start = iBrack; open = '[';
  } else if (iBrack === -1) {
    start = iBrace; open = '{';
  } else {
    start = Math.min(iBrace, iBrack);
    open = t[start] === '{' ? '{' : '[';
  }
  const close = open === '{' ? '}' : ']';
  let depth = 0;
  let end = -1;
  for (let i = start; i < t.length; i++) {
    if (t[i] === open) depth++;
    else if (t[i] === close) { depth--; if (depth === 0) { end = i; break; } }
  }
  if (end === -1) throw new Error('AI 返回的 JSON 括号不匹配');
  return JSON.parse(t.slice(start, end + 1));
}

function be32(buf, o) {
  return (buf[o] * 0x1000000 + buf[o + 1] * 0x10000 + buf[o + 2] * 0x100 + buf[o + 3]) >>> 0;
}
function be16(buf, o) {
  return buf[o] * 0x100 + buf[o + 1];
}

// ---- PalmDB 头解析 ----
export function parsePalmDB(buf) {
  const type = String.fromCharCode(buf[60], buf[61], buf[62], buf[63]);
  const creator = String.fromCharCode(buf[64], buf[65], buf[66], buf[67]);
  const numRec = be16(buf, 76);
  const recOff = [];
  for (let i = 0; i < numRec; i++) recOff.push(be32(buf, 78 + i * 8));
  return { type, creator, numRec, recOff };
}

// ---- Mobipocket 内置字节流解压缩（多字节安全） ----
// 直接对字节流操作：1-8=字面复制 c 字节；9-0x7F=字面字符(9=空格)；0x80-0xBF=后向引用；0xC0-0xFF=空格+字符
function mobipocketDecode(data) {
  const out = [];
  let i = 0;
  const n = data.length;
  while (i < n) {
    const c = data[i++];
    if (c === 0) break; // 0x00 结束
    if (c <= 8) {
      // 1-8：复制 c 个字面字节
      for (let k = 0; k < c && i < n; k++) out.push(data[i++]);
    } else if (c < 0x80) {
      out.push(c === 0x09 ? 0x20 : c);
    } else if (c < 0xc0) {
      // 0x80-0xBF：2 字节后向引用
      if (i >= n) break;
      const c2 = data[i++];
      const concat = (c << 8) | c2;
      const distance = (concat >> 3) & 0x7ff;
      let length = (concat & 7) + 3;
      let s = out.length - distance;
      if (s < 0) { length += s; s = 0; } // 越界则裁剪
      for (let k = 0; k < length; k++) out.push(s + k < out.length ? out[s + k] : 0x20);
    } else {
      // 0xC0-0xFF：空格 + 字符(b ^ 0x80)
      out.push(0x20);
      out.push(c ^ 0x80);
    }
  }
  return out;
}

// 把解码后的字节数组（可能含跨记录残余）按 UTF-8 安全转为字符串，去掉无效字节
function bytesToUtf8Safe(bytes) {
  // 去掉末尾不完整的多字节序列
  let end = bytes.length;
  while (end > 0 && bytes[end - 1] >= 0x80 && bytes[end - 1] < 0xc0) end--;
  let s = '';
  let i = 0;
  while (i < end) {
    const b = bytes[i];
    if (b < 0x80) { s += String.fromCharCode(b); i++; }
    else if (b >= 0xc0) {
      // 尝试按 UTF-8 解码一个码点
      let extra = 0;
      if (b >= 0xf0) extra = 3; else if (b >= 0xe0) extra = 2; else extra = 1;
      if (i + extra < end) {
        try {
          const slice = bytes.slice(i, i + extra + 1);
          s += new TextDecoder('utf-8').decode(Uint8Array.from(slice));
          i += extra + 1;
        } catch (e) { s += '?'; i++; }
      } else { s += '?'; i++; }
    } else { s += '?'; i++; }
  }
  return s;
}

// ---- EXTH 解析（取标题/作者等元数据） ----
function parseExth(buf, exthOff) {
  const out = {};
  if (buf[exthOff] !== 0x45 || buf[exthOff + 1] !== 0x58 || buf[exthOff + 2] !== 0x54 || buf[exthOff + 3] !== 0x48) return out;
  const exthLen = be32(buf, exthOff + 4);
  let p = exthOff + 8;
  const end = exthOff + exthLen;
  while (p + 8 <= end) {
    const type = be32(buf, p);
    const len = be32(buf, p + 4);
    if (len < 8) break;
    const data = buf.slice(p + 8, p + len);
    out[type] = data;
    p += len;
  }
  return out;
}

// 把第 pageIndex 页渲染为 JPEG dataURL；targetW 控制清晰度上限，避免吃内存
export async function renderPdfPage(doc, pageIndex, targetW) {
  const page = await doc.getPage(pageIndex + 1);
  const base = page.getViewport({ scale: 1 });
  const scale = targetW / base.width;
  const vp = page.getViewport({ scale });
  const canvas = document.createElement('canvas');
  canvas.width = Math.max(1, Math.ceil(vp.width));
  canvas.height = Math.max(1, Math.ceil(vp.height));
  const ctx = canvas.getContext('2d');
  ctx.fillStyle = '#ffffff';
  ctx.fillRect(0, 0, canvas.width, canvas.height);
  await page.render({ canvasContext: ctx, viewport: vp }).promise;
  const url = canvas.toDataURL('image/jpeg', 0.82);
  if (page.cleanup) page.cleanup();
  return { url, w: canvas.width, h: canvas.height };
}
